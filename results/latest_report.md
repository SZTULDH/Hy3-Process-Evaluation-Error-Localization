```python
def merge_intervals(intervals):
    if not intervals:
        return []
    intervals = sorted(intervals, key=lambda x: x[0])
```

```python
def length_of_longest_substring(s: str) -> int:
    if not s:
        return 0
    left = 0
    best = 0
    seen = set()
    for right, ch in enumerate(s):
        if ch in seen:
            seen.discard(s[left])
            left += 1
        seen.add(ch)
        best = max(best, right - left + 1)
    return best
```
